       for (let rank = 0; rank < vectorResults.length; rank++) {
         const id = vectorResults[rank].checkpoint_id;
         if (id) {
           const entry = scores.get(id) ?? { rrf: 0, vectorRank: null, bm25Rank: null };
           entry.rrf += 1 / (RRF_K + rank + 1);
           entry.vectorRank = rank + 1;
           scores.set(id, entry);
         }
       }
       for (let rank = 0; rank < bm25Results.length; rank++) {
         const id = bm25Results[rank].checkpoint_id;
         if (id) {
           const entry = scores.get(id) ?? { rrf: 0, vectorRank: null, bm25Rank: null };
           entry.rrf += 1 / (RRF_K + rank + 1);
           entry.bm25Rank = rank + 1;
           scores.set(id, entry);
         }
       }
       return [...scores.entries()]
         .sort((a, b) => b[1].rrf - a[1].rrf)
         .slice(0, limit)
         .map(([checkpoint_id, s]) => ({ checkpoint_id, ...s }));
     }
