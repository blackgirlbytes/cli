func TestEnrichWhyBlocks_AddsEntireMetadata(t *testing.T) {
	t.Parallel()

	blocks := []whyBlock{{
		StartLine: 23,
		EndLine:   24,
		Commit: whyCommitInfo{
			Hash: "b550507521111111111111111111111111111111",
		},
	}}

	origResolve := resolveWhyCommitMetadata
	t.Cleanup(func() { resolveWhyCommitMetadata = origResolve })

	resolveWhyCommitMetadata = func(_ context.Context, hash string) (whyCommitInfo, *whyEntireInfo, error) {
		return whyCommitInfo{
				Hash:      hash,
				ShortHash: "b550507521",
				Author:    "Victor Gutierrez",
				Subject:   "Handle interrupt signals",
			}, &whyEntireInfo{
				CheckpointID: "a3b2c4d5e6f7",
				Agent:        "Claude Code",
				Summary:      "Add signal handling for graceful CLI shutdown.",
				IsFallback:   false,
			}, nil
	}

	got, err := enrichWhyBlocks(context.Background(), blocks)
	if err != nil {
		t.Fatalf("enrich error: %v", err)
	}
	if got[0].Entire == nil {
		t.Fatal("expected Entire enrichment")
	}
	if got[0].Entire.Agent != "Claude Code" {
		t.Fatalf("agent = %q, want %q", got[0].Entire.Agent, "Claude Code")
	}
}
