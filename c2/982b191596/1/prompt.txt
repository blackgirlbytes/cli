Implementation seam shape:

```go
var runGitBlamePorcelain = func(ctx context.Context, path string) (string, error) {
	cmd := exec.CommandContext(ctx, "git", "blame", "--line-porcelain", "--", path)
	out, err := cmd.Output()
	if err != nil {
		return "", err
	}
	return string(out), nil
}

func loadWhyBlocks(ctx context.Context, path string) ([]whyBlock, error) {
	out, err := runGitBlamePorcelain(ctx, path)
	if err != nil {
		return nil, err
	}
	return parseWhyBlamePorcelain(strings.NewReader(out))
}
```
