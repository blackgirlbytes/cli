Planned command test:

```go
func TestWhyCommand_StaticModeRendersBlocks(t *testing.T) {
	t.Parallel()

	origLoad := loadWhyDisplayData
	t.Cleanup(func() { loadWhyDisplayData = origLoad })

	loadWhyDisplayData = func(_ context.Context, path string) ([]whyBlock, error) {
		if path != "cmd/entire/main.go" {
			t.Fatalf("path = %q, want %q", path, "cmd/entire/main.go")
		}
		return []whyBlock{{
			StartLine: 1,
			EndLine:   1,
			Commit:    whyCommitInfo{ShortHash: "7f93a963e3", Author: "Stefan Haubold", WhenText: "3 months ago"},
			CodeLines: []string{"package main"},
		}}, nil
	}

	cmd := newWhyCmd()
	var out bytes.Buffer
	cmd.SetOut(&out)
	cmd.SetErr(&out)
	cmd.SetArgs([]string{"cmd/entire/main.go"})

	if err := cmd.Execute(); err != nil {
		t.Fatalf("execute error: %v", err)
	}
	if !strings.Contains(out.String(), "package main") {
		t.Fatalf("output missing code block:\n%s", out.String())
	}
}
```
