```go
package cli

type whyCommitInfo struct {
	Hash      string
	ShortHash string
	Author    string
	Email     string
	WhenText  string
	Subject   string
}

type whyEntireInfo struct {
	CheckpointID string
	Agent        string
	Summary      string
	IsFallback   bool
}

type whyDetailKind string

const (
	whyDetailCheckpoint whyDetailKind = "checkpoint"
	whyDetailCommit     whyDetailKind = "commit"
)

type whyBlock struct {
	StartLine int
	EndLine   int
	Commit    whyCommitInfo
	Entire    *whyEntireInfo
	CodeLines []string
	Detail    whyDetailKind
}

func (b whyBlock) RangeText() string {
	if b.StartLine == b.EndLine {
		return itoa(b.StartLine)
	}
	return itoa(b.StartLine) + "-" + itoa(b.EndLine)
}
```

Also add a tiny local helper in the same file to avoid dragging formatting logic into tests early:

```go
func itoa(v int) string {
	return strconv.Itoa(v)
}
```
